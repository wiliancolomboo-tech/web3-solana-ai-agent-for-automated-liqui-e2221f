import { Connection, Keypair } from "@solana/web3.js";
import type { AppConfig, DexAdapter, PoolConfig } from "../types.js";
import { OrcaAdapter } from "./orcaAdapter.js";
import { RaydiumAdapter } from "./raydiumAdapter.js";
import { NoopAdapter } from "./noopAdapter.js";

export function createAdapter(cfg: AppConfig, pool: PoolConfig, conn: Connection): DexAdapter {
  if (cfg.executionMode === "dry-run") return new NoopAdapter(pool.dex);
  if (pool.dex === "orca") return new OrcaAdapter(pool.poolAddress, conn);
  if (pool.dex === "raydium") return new RaydiumAdapter(pool.poolAddress, conn);
  throw new Error(`Unsupported dex ${pool.dex}`);
}
