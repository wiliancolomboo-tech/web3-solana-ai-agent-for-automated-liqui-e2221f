import { Connection } from "@solana/web3.js";
import { loadConfig } from "./config.js";
import { StateStore } from "./storage.js";
import { Executor } from "./executor.js";
import { RebalanceMonitor } from "./monitor.js";
import pino from "pino";
// import bs58 from "bs58"; // use for private-key decode if using base58 secret

const log = pino({ level: process.env.LOG_LEVEL || "info" });

async function main() {
  const cfg = loadConfig();
  const store = new StateStore(cfg.dbPath);

  const conn = new Connection(cfg.rpcUrl, { commitment: "confirmed" });

  // For production, wire this from secure key mgmt (AWS KMS / Vault / Key Vault / HSM)
  const dummySecret = new Uint8Array(64).fill(1);
  // eslint-disable-next-line @typescript-eslint/no-var-requires
  const kp = require("@solana/web3.js").Keypair.fromSecretKey(dummySecret);

  const exec = new Executor(conn, kp, cfg.executionMode, cfg.maxRetries, cfg.maxSlippageBps);
  const monitor = new RebalanceMonitor(cfg, store, exec);

  monitor.start();
  log.info(`liquidity rebalancer started in ${cfg.executionMode}`);
}

main().catch((e) => {
  log.error(e, "fatal");
  process.exit(1);
});
