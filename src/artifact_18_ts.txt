import { describe, it, expect } from "vitest";
import { KineticIndicator } from "../indicators/kinetic";

describe("KineticIndicator", () => {
  it("returns stable value with increasing prices", () => {
    const k = new KineticIndicator(500);
    const base = [1, 1.01, 1.02, 1.03, 1.04, 1.05, 1.08, 1.12, 1.11, 1.10];
    let sig: any = null;
    for (const p of base) sig = k.push(p);
    expect(sig.ok).toBe(true);
    expect(sig.kineticScore).toBeGreaterThanOrEqual(0);
    expect(sig.velocity).toBeDefined();
  });

  it("handles insufficient data", () => {
    const k = new KineticIndicator();
    const a = k.push(100);
    const b = k.push(101);
    expect(a.ok).toBe(false);
    expect(b.ok).toBe(false);
  });
});
